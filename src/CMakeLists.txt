# Whole-program optimization for our own targets: the board calls its chips
# across source files every cycle.
include(CheckIPOSupported)
check_ipo_supported(RESULT RC_IPO LANGUAGES C)
if(RC_IPO)
    get_property(RC_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    set_target_properties(${RC_TARGETS} PROPERTIES
        INTERPROCEDURAL_OPTIMIZATION_RELEASE ON INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO ON)
endif()

# Profile-guided optimisation takes two builds: RC_PGO=generate, a training
# run (tools/pgo_train.py), then RC_PGO=use.
set(RC_PGO "" CACHE STRING "Profile-guided optimisation: generate or use")
if(RC_PGO STREQUAL "generate" OR RC_PGO STREQUAL "use")
    if(MSVC)
        # The profile belongs to the program: its code is generated when it links.
        if(RC_PGO STREQUAL "generate")
            target_link_options(rec64 PRIVATE /GENPROFILE)
        else()
            target_link_options(rec64 PRIVATE /USEPROFILE)
        endif()
    elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU")
        get_property(RC_PGO_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
        foreach(target ${RC_PGO_TARGETS})
            if(RC_PGO STREQUAL "generate")
                target_compile_options(${target} PRIVATE -fprofile-generate)
                target_link_options(${target} PRIVATE -fprofile-generate)
            else()
                target_compile_options(${target} PRIVATE -fprofile-use -fprofile-correction -Wno-missing-profile)
                target_link_options(${target} PRIVATE -fprofile-use)
            endif()
        endforeach()
    else()
        message(FATAL_ERROR "RC_PGO is supported with MSVC and GCC")
    endif()
elseif(RC_PGO)
    message(FATAL_ERROR "RC_PGO is generate or use")
endif()
