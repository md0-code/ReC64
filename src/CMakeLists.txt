cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" RC_VERSION_TEXT)
string(STRIP "${RC_VERSION_TEXT}" RC_VERSION_TEXT)
project(ReC64 VERSION ${RC_VERSION_TEXT} LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(cmake/Warnings.cmake)
configure_file(rc_version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/rc_version.h @ONLY)
get_filename_component(RC_WORK "${CMAKE_CURRENT_SOURCE_DIR}/../work" ABSOLUTE)
file(GLOB RC_YAML_SOURCES CONFIGURE_DEPENDS third_party/libyaml/src/*.c)
add_library(rc_yaml STATIC ${RC_YAML_SOURCES})
target_include_directories(rc_yaml PUBLIC third_party/libyaml/include)
target_compile_definitions(rc_yaml PUBLIC YAML_DECLARE_STATIC PRIVATE
    YAML_VERSION_MAJOR=0 YAML_VERSION_MINOR=2 YAML_VERSION_PATCH=5 YAML_VERSION_STRING="0.2.5" _CRT_SECURE_NO_WARNINGS)
if(WIN32)
    add_library(rc_host STATIC host/host_win32.c)
    target_link_libraries(rc_host PUBLIC ws2_32)
else()
    add_library(rc_host STATIC host/host_posix.c)
endif()
rc_set_warnings(rc_host)
add_library(rc_core STATIC core/config.c core/yamledit.c core/romdb.c core/state.c
    core/media.c core/font8x8.c core/stick.c core/machine.c core/stub_machine.c hw/sched.c)
target_include_directories(rc_core PUBLIC core hw PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(rc_core PUBLIC rc_yaml rc_host)
if(NOT WIN32)
    target_link_libraries(rc_core PUBLIC m)
endif()
rc_set_warnings(rc_core)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL2 URL https://www.libsdl.org/release/SDL2-2.32.10.tar.gz
    URL_HASH SHA256=5f5993c530f084535c65a6879e9b26ad441169b3e25d789d83287040a9ca5165
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL2)
add_library(rc_shell STATIC app/harness.c app/pace.c)
target_link_libraries(rc_shell PUBLIC rc_core)
rc_set_warnings(rc_shell)
add_library(rc_platform STATIC platform/video.c platform/audio.c platform/audiosync.c platform/png.c platform/overlay.c)
target_link_libraries(rc_platform PUBLIC rc_shell SDL2::SDL2-static)
target_compile_definitions(rc_platform PUBLIC SDL_MAIN_HANDLED)
rc_set_warnings(rc_platform)
add_executable(rec64 app/rec64.c app/desktop.c)
target_include_directories(rec64 PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(rec64 PRIVATE rc_platform)
rc_set_warnings(rec64)
enable_testing()
add_executable(test_foundation tests/test_foundation.c)
target_link_libraries(test_foundation PRIVATE rc_core)
rc_set_warnings(test_foundation)
add_test(NAME foundation COMMAND test_foundation)
